        function loadOrdersFromStorage() {
            const stored = localStorage.getItem('app_orders_data');
            if (stored) {
                try {
                    orders = JSON.parse(stored);
                } catch(e) {
                    orders = initialSampleOrders;
                }
            } else {
                orders = initialSampleOrders;
                saveOrdersToStorage();
            }
        }

        function saveOrdersToStorage() {
            localStorage.setItem('app_orders_data', JSON.stringify(orders));
        }

        // Render UI
        function renderOrders() {
            const grid = document.getElementById('orders-grid');
            const emptyState = document.getElementById('empty-state');
            const searchVal = document.getElementById('search-input').value.trim().toLowerCase();

            grid.innerHTML = '';

            // Filter logic
            const filtered = orders.filter(order => {
                const matchesFilter = currentFilter === 'all' || order.status === currentFilter;
                const matchesSearch = order.customerName.toLowerCase().includes(searchVal) || 
                                      order.phone.includes(searchVal) ||
                                      order.productName.toLowerCase().includes(searchVal);
                return matchesFilter && matchesSearch;
            });

            // Update Statistics
            updateStats();

            if (filtered.length === 0) {
                grid.classList.add('hidden');
                emptyState.classList.remove('hidden');
                return;
            }

            grid.classList.remove('hidden');
            emptyState.classList.add('hidden');

            // Build Cards
            filtered.forEach(order => {
                const totalAmount = order.price * order.quantity;
                const card = document.createElement('div');

                // Status badge styling

                        statusBadge = '<span class="px-3 py-1 bg-emerald-50 text-emerald-600 border border-emerald-200 rounded-full text-xs font-bold">تم التسليم</span>';
                        break;
                    case 'ملغي':
                        statusBadge = '<span class="px-3 py-1 bg-rose-50 text-rose-600 border border-rose-200 rounded-full text-xs font-bold">ملغي</span>';

                        <div class="relative h-48 bg-slate-100 overflow-hidden group">
                            <img src="${order.image || 'https://placehold.co/400x300/f1f5f9/94a3b8?text=لا+توجد+صورة'}" class="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500">

                            </div>
                        </div>

                            <!-- Product Name & Price -->
                            <div class="flex justify-between items-start">
                                <div>

                                <div class="text-left">

                                    <span class="text-base font-extrabold text-sky-600">${totalAmount.toLocaleString()} د.ج</span>
                                </div>
                            </div>

                            <!-- Customer Info -->
                            <div class="space-y-2 text-xs text-slate-600">

                        <!-- Status Quick Switcher -->
                        <select onchange="updateOrderStatus('${order.id}', this.value)" class="text-xs font-semibold py-1.5 px-2.5 rounded-lg border border-slate-200 bg-white focus:outline-none">
                            <option value="معلق" ${order.status === 'معلق' ? 'selected' : ''}>قيد الانتظار</option>
                            <option value="جاري التجهيز" ${order.status === 'جاري التجهيز' ? 'selected' : ''}>جاري التجهيز</option>

        // Stats Calculation
        function updateStats() {
            const total = orders.length;
            const pending = orders.filter(o => o.status === 'معلق' || o.status === 'جاري التجهيز').length;
            const completed = orders.filter(o => o.status === 'تم التسليم').length;
            const revenue = orders
                .filter(o => o.status === 'تم التسليم')
                .reduce((sum, o) => sum + (o.price * o.quantity), 0);

            document.getElementById('stat-total').innerText = total;
            document.getElementById('stat-pending').innerText = pending;
            document.getElementById('stat-completed').innerText = completed;
            document.getElementById('stat-revenue').innerText = revenue.toLocaleString() + ' د.ج';
        }

        // Filters
        function setFilter(filterType) {
            currentFilter = filterType;
            document.querySelectorAll('.filter-btn').forEach(btn => {
                btn.classList.remove('bg-sky-500', 'text-white', 'shadow-sm');
                btn.classList.add('bg-slate-100', 'text-slate-600');
            });

            const activeBtn = document.getElementById(`filter-${filterType}`);
            if (activeBtn) {
                activeBtn.classList.remove('bg-slate-100', 'text-slate-600');
                activeBtn.classList.add('bg-sky-500', 'text-white', 'shadow-sm');
            }
            renderOrders();
        }

        // Image Base64 Handler
        function handleImageUpload(e) {
            const file = e.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(evt) {
                    currentImageBase64 = evt.target.result;
                    document.getElementById('image-preview').src = currentImageBase64;
                };
                reader.readAsDataURL(file);
            }
        }

        // Add Order Handler
        function handleSaveOrder(e) {
            e.preventDefault();

            const newOrder = {
                id: 'ORD-' + Math.floor(1000 + Math.random() * 9000),
                customerName: document.getElementById('cust-name').value.trim(),
                phone: document.getElementById('cust-phone').value.trim(),
                address: document.getElementById('cust-address').value.trim(),
                productName: document.getElementById('prod-name').value.trim(),
                price: parseFloat(document.getElementById('prod-price').value) || 0,
                quantity: parseInt(document.getElementById('prod-qty').value) || 1,
                status: document.getElementById('order-status').value,
                image: currentImageBase64 || 'https://placehold.co/400x300/f1f5f9/94a3b8?text=منتج+بدون+صورة',
                notes: document.getElementById('order-notes').value.trim(),
                date: new Date().toISOString().split('T')[0]
            };

            orders.unshift(newOrder);
            saveOrdersToStorage();
            renderOrders();
            closeModal();
            resetForm();
        }

        // Quick Status Update
        function updateOrderStatus(id, newStatus) {
            const index = orders.findIndex(o => o.id === id);
            if (index !== -1) {
                orders[index].status = newStatus;
                saveOrdersToStorage();
                renderOrders();
            }
        }

        // Delete Order
        function deleteOrder(id) {
            if (confirm('هل أنت تأكد من رغبتك في حذف هذا الطلب؟')) {
                orders = orders.filter(o => o.id !== id);
                saveOrdersToStorage();
                renderOrders();
            }
        }

        // Print Receipt Function
        function printReceipt(id) {
            const order = orders.find(o => o.id === id);
            if (!order) return;

            document.getElementById('rec-date').innerText = 'تاريخ الإصدار: ' + (order.date || new Date().toISOString().split('T')[0]);
            document.getElementById('rec-id').innerText = order.id;
            document.getElementById('rec-cust-name').innerText = order.customerName;
            document.getElementById('rec-cust-phone').innerText = order.phone;
            document.getElementById('rec-cust-address').innerText = order.address;
            
            document.getElementById('rec-prod-name').innerText = order.productName;
            document.getElementById('rec-prod-price').innerText = order.price.toLocaleString() + ' د.ج';
            document.getElementById('rec-prod-qty').innerText = order.quantity;
            document.getElementById('rec-prod-total').innerText = (order.price * order.quantity).toLocaleString() + ' د.ج';

            if (order.notes) {
                document.getElementById('rec-notes-container').style.display = 'block';
                document.getElementById('rec-notes').innerText = order.notes;
            } else {
                document.getElementById('rec-notes-container').style.display = 'none';
            }

            window.print();
        }

        // Export Data (Backup)
        function exportData() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(orders, null, 2));
            const downloadAnchor = document.createElement('a');
            downloadAnchor.setAttribute("href", dataStr);
            downloadAnchor.setAttribute("download", `orders_backup_${new Date().toISOString().split('T')[0]}.json`);
            document.body.appendChild(downloadAnchor);
            downloadAnchor.click();
            downloadAnchor.remove();
        }

        // Modal Helpers
        function openModal() {
            document.getElementById('order-modal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('order-modal').classList.add('hidden');
            resetForm();
        }

        function resetForm() {
            document.getElementById('add-order-form').reset();
            currentImageBase64 = '';
            document.getElementById('image-preview').src = 'https://placehold.co/300x300/f1f5f9/94a3b8?text=صورة+المنتج';
        }

        function openImageModal(imgSrc) {
            if (!imgSrc) return;
            document.getElementById('full-image-target').src = imgSrc;
            document.getElementById('full-image-modal').classList.remove('hidden');
        }

        function closeImageModal() {
            document.getElementById('full-image-modal').classList.add('hidden');
        }

        function escapeHtml(text) {
            if (!text) return '';
            return text.replace(/&/g, "&amp;")
                       .replace(/</g, "&lt;")
                       .replace(/>/g, "&gt;")
                       .replace(/"/g, "&quot;")
                       .replace(/'/g, "&#039;");
        }
